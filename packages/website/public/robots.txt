# them yields an unbounded set of URLs whose server-rendered shell has no
# content, which Google reports as soft 404s. The routes themselves (/text/,
# /url/, /explore/) stay indexable — only the parameterized forms are blocked.
Disallow: /*?text=
Disallow: /*?url=
Disallow: /*?word=
Disallow: /*?lang=
